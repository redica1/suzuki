function releaseNow() { release(); }
function scheduleRelease() {
  need_(PROD);
  const m = String(RELEASE_AT).match(/^(\d{4})-(\d{2})-(\d{2})\s+(\d{1,2}):(\d{2})$/);
  if (!m) throw new Error("RELEASE_AT 에 '2026-12-01 02:00' 처럼 시각을 적어 주세요.");
  const at = new Date(m[1] + '-' + m[2] + '-' + m[3] + 'T' + ('0' + m[4]).slice(-2) + ':' + m[5] + ':00+09:00');
  if (at.getTime() < Date.now() + 60000) throw new Error('지금보다 나중 시각을 적어 주세요.');
  cancelRelease(); ScriptApp.newTrigger('release').timeBased().at(at).create(); P_().setProperty('releaseAt', RELEASE_AT);
  Logger.log('예약했어요: ' + RELEASE_AT);
}
function cancelRelease() { ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'release').forEach(t => ScriptApp.deleteTrigger(t)); P_().deleteProperty('releaseAt'); }

function release() {
  need_(PROD);
  const lock = LockService.getScriptLock(); if (!lock.tryLock(60000)) throw new Error('다른 반영이 진행 중이에요.');
  const log = [];
  try {
    const f = DriveApp.getFileById(PROD.sheetId);
    const b = f.makeCopy('[백업] ' + f.getName() + ' ' + now_());
    log.push('메인 시트 백업: ' + b.getName() + ' (' + b.getUrl() + ')');
    const prev = currentVersion_(PROD);
    const src = fetchSrc_();
    if (P_().getProperty('hash_stg') && src.hash !== P_().getProperty('hash_stg')) log.push('주의: 깃허브 코드가 마지막 STG 반영 뒤에 또 바뀌었어요. (그래도 지금 코드로 반영)');
    const v = deployFiles_(PROD, src, '메인 반영 (STG 확인 코드)');
    P_().setProperty('prevVersion', String(prev));
    log.push('메인 앱 버전 ' + prev + ' → ' + v + ' (같은 주소)');
  } catch (e) { log.push('실패: ' + e.message); throw e; }
  finally {
    lock.releaseLock(); cancelRelease();
    const body = log.join('\n') + '\n\n되돌리려면 도우미에서 rollback 을 실행하세요.';
    Logger.log(body);
    try { MailApp.sendEmail(Session.getEffectiveUser().getEmail(), '[캠프 접수] 메인 반영 결과 ' + now_(), body); } catch (x) {}
  }
}
function rollback() {
  const prev = +P_().getProperty('prevVersion'); if (!prev) throw new Error('되돌릴 이전 버전 기록이 없어요.');
  const cur = currentVersion_(PROD);
  api_(PROD, 'put', '/deployments/' + PROD.deploymentId, { deploymentConfig: { versionNumber: prev, manifestFileName: 'appsscript', description: '되돌림 ' + now_() } });
  Logger.log('메인 앱을 버전 ' + cur + ' → ' + prev + ' 로 되돌렸어요.');
}

  out.push('예약: ' + (P_().getProperty('releaseAt') || '없음') + ' / 되돌릴 버전: ' + (P_().getProperty('prevVersion') || '없음'));
  Logger.log(out.join('\n'));
}

function need_(t) { if (!t.scriptId || !t.deploymentId) throw new Error(t.name + ' 의 scriptId 를 먼저 채워 주세요.'); }

/** 깃허브에서 files.json 에 적힌 파일을 모두 받아 옴.
 *  · 같은 커밋(sha)에서 받음 → 올리는 도중에 섞인 파일을 받지 않음
 *  · 파일마다 적힌 빌드 번호가 다르면 이번에는 건너뛰고 5분 뒤 다시 (MIXED) */
function srcBase_(srcUrl) {
  const SRC0 = srcUrl || SRC;
  try {
    const r = UrlFetchApp.fetch('https://api.github.com/repos/redica1/suzuki/commits/main', { headers: { Accept: 'application/vnd.github.sha' }, muteHttpExceptions: true });
    const sha = r.getContentText().trim();
    if (r.getResponseCode() === 200 && /^[0-9a-f]{40}$/.test(sha)) return SRC0.replace('/main/', '/' + sha + '/');
  } catch (e) {}
  return SRC0;
}
function fetchSrc_(srcUrl) {
  const base = srcBase_(srcUrl);
  const get = n => { const r = UrlFetchApp.fetch(base + n + '?t=' + Date.now(), { muteHttpExceptions: true });
    if (r.getResponseCode() !== 200) throw new Error('깃허브에서 ' + n + ' 를 못 받았어요 (' + r.getResponseCode() + ')'); return r.getContentText('UTF-8'); };
  const names = JSON.parse(get('files.json'));
  const files = names.map(n => ({ n, src: get(n) }));
  if (!files.some(f => f.n === 'Code.js') || !files.some(f => f.n === 'Index.html')) throw new Error('Code.js · Index.html 이 없어요.');
  const builds = {}; files.forEach(f => { const m = f.src.slice(0, 3000).match(/build: (\S+)/); if (m) builds[m[1]] = 1; });
  if (Object.keys(builds).length > 1) throw new Error('MIXED 깃허브 파일이 아직 다 바뀌지 않았어요 (빌드 ' + Object.keys(builds).join(', ') + '). 5분 뒤 다시 해요.');
  const hash = Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, files.map(f => f.n + '\n' + f.src).join('\n\u0000\n'), Utilities.Charset.UTF_8));
  return { files, hash, build: Object.keys(builds)[0] || '' };
}

function api_(t, method, path, body) {
  const res = UrlFetchApp.fetch('https://script.googleapis.com/v1/projects/' + t.scriptId + path, {
    method, contentType: 'application/json; charset=utf-8', payload: body ? JSON.stringify(body).replace(/[\u007f-\uffff]/g, c => '\\u' + ('000' + c.charCodeAt(0).toString(16)).slice(-4)) : undefined,   // 한글·특수 글자는 \uXXXX 로 보내 깨지지 않게
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() }, muteHttpExceptions: true });
  const code = res.getResponseCode(), txt = res.getContentText();
  if (code >= 300) throw new Error(t.name + ' ' + path + ' 실패 (' + code + '): ' + txt.slice(0, 400));
  return txt ? JSON.parse(txt) : {};
}
function currentVersion_(t) { const d = api_(t, 'get', '/deployments/' + t.deploymentId); return (d.deploymentConfig && d.deploymentConfig.versionNumber) || 0; }

/** 앱의 appsscript.json(액세스 「모든 사용자」 설정)은 그대로 두고, 코드 파일만 깃허브 것으로 바꾼 뒤 새 버전으로 같은 주소에 배포 */
function deployFiles_(t, src, why) {
  const cur = api_(t, 'get', '/content').files || [];
  const manifest = cur.filter(f => f.type === 'JSON');
  if (!manifest.length || !/ANYONE_ANONYMOUS/.test(manifest[0].source)) throw new Error(t.name + ' 의 appsscript.json 액세스가 「모든 사용자」(ANYONE_ANONYMOUS)가 아니에요. 먼저 고쳐 주세요.');
  const put = src.files.map(f => ({ name: f.n.replace(/\.[^.]+$/, ''), type: typeOf_(f.n), source: f.src }));
  api_(t, 'put', '/content', { files: manifest.concat(put) });
  const when = now_();
  const v = api_(t, 'post', '/versions', { description: why + ' ' + when });
  api_(t, 'put', '/deployments/' + t.deploymentId, { deploymentConfig: { versionNumber: v.versionNumber, manifestFileName: 'appsscript', description: why + ' ' + when } });
  Logger.log(t.name + ' 버전 ' + v.versionNumber + ' 배포 완료 (' + why + ', ' + when + ')');
  return v.versionNumber;
}
